#   Install torch FIRST:
#       pip install torch --index-url https://download.pytorch.org/whl/cu128
#   If the stable cu128 wheel lacks sm_120, use the nightly:
#       pip install --pre torch --index-url https://download.pytorch.org/whl/nightly/cu128
#
#   Verify:
#       python -c "import torch; print(torch.cuda.get_arch_list())"
#
#   Then:  pip install -r requirements.txt
# =====================================================================

scikit-learn==1.6.1         # training only — see note below

tqdm

# ---------------------------------------------------------------------
# scikit-learn is needed to TRAIN (StandardScaler is fitted during
# training) but not to PREDICT. The released bundle stores the scaler as
# plain numpy arrays, so predict.py applies (X - mean) / scale directly
# and never unpickles a scikit-learn estimator. Inference therefore works
# under any scikit-learn, or none at all.
#
# If you retrain, run scripts/scaler_to_arrays.py on the new bundle so it
# carries the arrays too. Without them predict.py falls back to the
# pickled scaler, which reintroduces the version coupling.
# ---------------------------------------------------------------------
